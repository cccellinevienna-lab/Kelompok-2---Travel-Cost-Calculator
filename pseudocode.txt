DEKLARASI VARIABEL:
    jarak            : REAL
    efisiensi        : REAL
    harga            : REAL
    jenisPerjalanan  : INTEGER
    totalJarak       : REAL
    konsumsiBBM      : REAL
    totalBiaya       : REAL
    ulang            : STRING

DO
    OUTPUT "Masukkan jarak perjalanan (km): "
    INPUT jarak

    OUTPUT "Masukkan efisiensi bahan bakar kendaraan (km/liter): "
    INPUT efisiensi

    OUTPUT "Masukkan harga bahan bakar per liter: "
    INPUT harga

    DO
        OUTPUT "Pilih jenis perjalanan (1 = Sekali Jalan, 2 = Pulang-Pergi): "
        INPUT jenisPerjalanan
        IF jenisPerjalanan != 1 AND jenisPerjalanan != 2 THEN
            OUTPUT "Pilihan tidak valid! Harus ketik angka [1] atau [2]."
        ENDIF
    UNTIL jenisPerjalanan != 1 AND jenisPerjalanan != 2

    IF jenisPerjalanan == 2 THEN
        totalJarak = jarak * 2
    ELSE
        totalJarak = jarak
    ENDIF

    konsumsiBBM = totalJarak / efisiensi
    totalBiaya  = konsumsiBBM * harga

    OUTPUT "Total Jarak Perjalanan: " + totalJarak + " km"
    OUTPUT "Estimasi Konsumsi BBM: " + konsumsiBBM + " liter"
    OUTPUT "Estimasi Total Biaya BBM: Rp" + totalBiaya

    DO
        OUTPUT "Apakah ingin menghitung lagi? (y/n): "
        INPUT ulang
        IF ulang != "y" AND ulang != "n" THEN
            OUTPUT "Pilihan tidak valid! Harus ketik huruf [y] atau [n]."
        ENDIF
    UNTIL ulang != "y" AND ulang != "n"

UNTIL ulang == "y"

END
